    let audioCtx = null;

    let audioEnabled = true;


    function playTone(freq = 880, type = 'sine', duration = 0.08) {

      if (!audioEnabled) return;

      try {

        if (!audioCtx) {

          audioCtx = new (window.AudioContext || window.webkitAudioContext)();

        }

        if (audioCtx.state === 'suspended') {

          audioCtx.resume();

        }

        const osc = audioCtx.createOscillator();

        const gain = audioCtx.createGain();

        osc.type = type;

        osc.frequency.setValueAtTime(freq, audioCtx.currentTime);

        gain.gain.setValueAtTime(0.08, audioCtx.currentTime);

        gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + duration);

        osc.connect(gain);

        gain.connect(audioCtx.destination);

        osc.start();

        osc.stop(audioCtx.currentTime + duration);

      } catch (e) {}

    }

    function exportAuditCSV() {

      const trades = getAuditTrades();

      if (trades.length === 0) {

        alert('No audit trades to export.');

        return;

      }


      const startingBal = parseFloat(localStorage.getItem('rjs_desk_start_bal')) || 10000;

      let runningBal = startingBal;


      let csv = 'Index,Date_Time,Pair,Position,Position_Value,Lot_Size,Stop_Loss,Take_Profit,Result,RRR,PnL_PHP,Running_Balance_PHP,Discipline_Notes\n';

      trades.forEach((t, i) => {

        runningBal += (t.pnl || 0);

        const cleanNote = (t.note || '').replace(/"/g, '""');

        const pos = t.position === 'SELL' || t.direction === 'SHORT' ? 'SELL' : 'BUY';

        const entryVal = t.entryPrice || t.positionValue || '-';

        const time = t.executionTime || t.timestamp || '09:30 PM PHT';

        const res = t.result || t.outcome || 'TP 1:3';

        const rVal = t.rrr || (String(res).includes('1:3') ? '1:3' : String(res).includes('1:1') ? '1:1' : String(res).includes('BE') ? '0:0' : '-1:1');

        csv += `${i + 1},"${time}","${t.pair || 'USTECH100'}","${pos}","${entryVal}",${Number(t.lotSize ?? 1.0).toFixed(2)},"${t.stopLoss || '-'}","${t.takeProfit || '-'}","${res}","${rVal}",${t.pnl || 0},${runningBal},"${cleanNote}"\n`;

      });


      const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });

      const url = URL.createObjectURL(blob);

      const link = document.createElement('a');

      link.setAttribute('href', url);

      link.setAttribute('download', `RyanJade_Phase2_AuditLedger_${new Date().toISOString().slice(0,10)}.csv`);

      document.body.appendChild(link);

      link.click();

      document.body.removeChild(link);

      playTone(700, 'sine', 0.05);

    }


    function refreshExecutionTimestamp() {

      const now = new Date();

      const utc = now.getTime() + (now.getTimezoneOffset() * 60000);

      const manilaTime = new Date(utc + (3600000 * 8));

      const hours = manilaTime.getHours();

      const minutes = manilaTime.getMinutes();

      const ampm = hours >= 12 ? 'PM' : 'AM';

      const formattedHours = String(hours % 12 || 12).padStart(2, '0');

      const formattedMinutes = String(minutes).padStart(2, '0');

      const timeStr = `${formattedHours}:${formattedMinutes} ${ampm} PHT`;

      

      const input = document.getElementById('tradeExecutionTime');

      if (input) {

        input.value = timeStr;

      }

    }
